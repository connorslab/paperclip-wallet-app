cmake_minimum_required(VERSION 3.22.1)
project(paperclip_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/${ANDROID_ABI}/libpaperclip_mobile.a")
if(NOT EXISTS "${RUST_LIBRARY}")
    message(FATAL_ERROR "Build the shared Rust engine first: mobile/android/scripts/build-native.sh")
endif()
add_library(paperclip_core STATIC IMPORTED)
set_target_properties(paperclip_core PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}")
add_library(paperclip_android SHARED bridge.cpp)
target_link_libraries(paperclip_android PRIVATE paperclip_core android log dl m)
target_link_options(paperclip_android PRIVATE "-Wl,-z,max-page-size=16384")
